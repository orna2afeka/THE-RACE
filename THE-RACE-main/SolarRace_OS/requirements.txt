# OPTIONAL — only for the standalone engineering dashboard (`python

# pyqtgraph>=0.13.3

# GPS needs NO pip packages. modules/gps_reader.py is a gpsd client speaking
# gpsd's JSON protocol over a plain stdlib socket (the same stream as
# `gpspipe -w`), so pyserial/pynmea2 are no longer used by anything here.
# GPS is a SYSTEM dependency instead — install and configure it on the Pi:
#   sudo apt install gpsd gpsd-clients
#   sudo nano /etc/default/gpsd     # DEVICES="/dev/ttyUSB1", GPSD_OPTIONS="-n"
#   sudo systemctl enable --now gpsd
# Verify with:  gpspipe -w        (or: python3 SolarRace_OS/modules/gps_reader.py)
