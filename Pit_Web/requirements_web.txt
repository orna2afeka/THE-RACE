# requirements_web.txt — everything this folder needs.
#
#     pip install -r Pit_Web/requirements_web.txt
#
# The pit dashboard (React/FastAPI) and the collector. Nothing here imports
# Streamlit; the speed-profile builder adds it on top from
# Pit_Dashboard/requirements_profiles.txt. The car has its own list in
# SolarRace_OS/requirements.txt.
#
# Two processes:
#     python Pit_Dashboard/collector.py             # Firebase -> telemetry.db
#     python -m uvicorn Pit_Web.api:app --port 8000 # the dashboard
#
# The pandas/numpy/matplotlib/openpyxl pins are the versions the pit laptop has
# run on; move them deliberately, not as a side effect.

# --- the web backend --------------------------------------------------------
fastapi>=0.115
pydantic>=2.9
uvicorn>=0.30
websockets>=13

# --- shared modules the backend imports -------------------------------------
# pandas/numpy  : strategy_engine, weather_service, the CSV export frame
# matplotlib    : imported at the top of strategy_engine.py. The dashboard
#                 no longer renders a PNG (the chart is drawn in the browser from the
#                 served simulation), but the import still has to resolve.
# openpyxl      : export.py's 3-sheet workbook
pandas==2.2.1
numpy==1.26.4
matplotlib==3.8.3
openpyxl==3.1.2

# --- collector --------------------------------------------------------------
# requests   : the RTDB REST stream, and weather_service
# google-auth: the collector's OAuth2 token from the service account
#
# requests is >= rather than == on purpose. The Streamlit version pinned 2.31.0,
# but the pit laptop already has a newer one, and a hard pin would make every
# launch downgrade a package other things share — for a plain HTTP GET with a
# timeout, which any recent version does identically. The four pins above stay
# exact because they shape OUTPUT: the workbook, the CSV and the strategy PNG.
requests>=2.31.0
google-auth==2.29.0

# Timezone database for zoneinfo. Windows ships no system tz database, so
# without this the export cannot resolve Asia/Jerusalem or Europe/Brussels and
# raises ZoneInfoNotFoundError. The pit laptop is Windows.
tzdata==2026.3
